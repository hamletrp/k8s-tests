            "--cluster-name=<your-cluster-name>",
            "--cluster-endpoint=https://<your-cluster-endpoint>",

  github actios :
    steps:
      - uses: actions/checkout@v3

      - name: Update image tag for app1 and app2
        run: |
          yq e '.app1.image.tag = "v1.2.3"' -i values.yaml
          yq e '.app2.image.tag = "v2.4.6"' -i values.yaml

      - name: Commit & push
        run: |
          git config user.name "ci-bot"
          git config user.email "ci@bot.local"
          git commit -am "ci: bump image tags"
          git push
