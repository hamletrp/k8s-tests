karpenter
  ServiceAccount
    # === Patch the service account ===
    kubectl annotate serviceaccount \
      karpenter \
      -n karpenter \
      eks.amazonaws.com/role-arn="arn:aws:iam::722249351142:role/KarpenterControllerRole-cluster-lab-13" \
      --overwrite

    kubectl patch serviceaccount karpenter -n karpenter \
    -p '{"metadata": {"annotations": {"eks.amazonaws.com/role-arn": "role-arn"}}}'

  Deployment
    kubectl -n karpenter patch deployment karpenter \
      --type='json' \
      -p='[
        {
          "op": "replace",
          "path": "/spec/template/spec/containers/0/args",
          "value": [
            "--cluster-name=<your-cluster-name>",
            "--cluster-endpoint=https://<your-cluster-endpoint>",
            "--aws-interruption-queue-name=<your-interruption-queue>"
          ]
        }
      ]'

    check:
      kubectl -n karpenter get deployment karpenter -o jsonpath='{.spec.template.spec.containers[0].args}'

      - name: Commit & push
        run: |
          git config user.name "ci-bot"
